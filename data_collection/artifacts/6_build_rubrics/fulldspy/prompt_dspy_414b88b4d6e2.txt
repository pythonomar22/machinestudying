```python
import dspy
from dspy.teleprompt import BootstrapFewShot
from dspy.utils.dummies import DummyLM


class SimpleModule(dspy.Module):
    def __init__(self):
        super().__init__()
        self.predictor = dspy.Predict("question -> answer")

    def forward(self, question):
        return self.predictor(question=question)


def tiny_nonzero_metric(example, prediction, trace=None):
    return 0.1


dspy.configure(lm=DummyLM({"math": {"answer": "4"}}))
trainset = [dspy.Example(question="math", answer="4").with_inputs("question")]

compiled_without_threshold = BootstrapFewShot(
    metric=tiny_nonzero_metric,
    max_bootstrapped_demos=1,
    max_labeled_demos=0,
).compile(SimpleModule(), teacher=SimpleModule(), trainset=trainset)

compiled_with_threshold = BootstrapFewShot(
    metric=tiny_nonzero_metric,
    metric_threshold=0.5,
    max_bootstrapped_demos=1,
    max_labeled_demos=0,
).compile(SimpleModule(), teacher=SimpleModule(), trainset=trainset)

count_without_threshold = len(compiled_without_threshold.predictor.demos)
count_with_threshold = len(compiled_with_threshold.predictor.demos)
print(count_without_threshold, count_with_threshold)
assert count_without_threshold == 1
assert count_with_threshold == 0
```

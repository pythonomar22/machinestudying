0511: def _convert_chat_request_to_responses_request(request: dict[str, Any]):
0512:     """
0513:     Convert a chat request to a responses request
0514:     See https://platform.openai.com/docs/api-reference/responses/create for the responses API specification.
0515:     Also see https://platform.openai.com/docs/api-reference/chat/create for the chat API specification.
0516:     """
0517:     request = dict(request)
0518:     if "messages" in request:
0519:         content_blocks = []
0520:         for msg in request.pop("messages"):
0521:             c = msg.get("content")
0522:             if isinstance(c, str):
0523:                 content_blocks.append({"type": "input_text", "text": c})
0524:             elif isinstance(c, list):
0525:                 # Convert each content item from Chat API format to Responses API format
0526:                 for item in c:
0527:                     content_blocks.append(_convert_content_item_to_responses_format(item))
0528:         request["input"] = [{"role": msg.get("role", "user"), "content": content_blocks}]
0529:     # Convert `reasoning_effort` to reasoning format supported by the Responses API
0530:     if "reasoning_effort" in request:

0537:         if isinstance(response_format, type) and issubclass(response_format, pydantic.BaseModel):
0538:             response_format = {

0540:                 "type": "json_schema",
0541:                 "schema": response_format.model_json_schema(),
0542:             }

0362: def encode_sft_example(example, tokenizer, max_seq_length):
0363:     """
0364:     This function encodes a single example into a format that can be used for sft training.
0365:     Here, we assume each example has a 'messages' field. Each message in it is a dict with 'role' and 'content' fields.
0366:     We use the `apply_chat_template` function from the tokenizer to tokenize the messages and prepare the input and label tensors.
0367: 
0368:     Code obtained from the allenai/open-instruct repository: https://github.com/allenai/open-instruct/blob/4365dea3d1a6111e8b2712af06b22a4512a0df88/open_instruct/finetune.py
0369:     """
0370:     import torch
0371: 
0372:     messages = example["messages"]
0373:     if len(messages) == 0:
0374:         raise ValueError("messages field is empty.")
0375:     input_ids = tokenizer.apply_chat_template(
0376:         conversation=messages,
0377:         tokenize=True,
0378:         return_tensors="pt",
0379:         padding=False,
0380:         truncation=True,
0381:         max_length=max_seq_length,
0382:         add_generation_prompt=False,
0383:     )
0384:     labels = input_ids.clone()
0385:     # mask the non-assistant part for avoiding loss
0386:     for message_idx, message in enumerate(messages):
0387:         if message["role"] != "assistant":
0388:             # we calculate the start index of this non-assistant message
0389:             if message_idx == 0:
0390:                 message_start_idx = 0
0391:             else:
0392:                 message_start_idx = tokenizer.apply_chat_template(
0393:                     conversation=messages[:message_idx],  # here marks the end of the previous messages
0394:                     tokenize=True,
0395:                     return_tensors="pt",
0396:                     padding=False,
0397:                     truncation=True,
0398:                     max_length=max_seq_length,
0399:                     add_generation_prompt=False,
0400:                 ).shape[1]
0401:             # next, we calculate the end index of this non-assistant message
0402:             if message_idx < len(messages) - 1 and messages[message_idx + 1]["role"] == "assistant":
0403:                 # for intermediate messages that follow with an assistant message, we need to
0404:                 # set `add_generation_prompt=True` to avoid the assistant generation prefix being included in the loss
0405:                 # (e.g., `<|assistant|>`)
0406:                 message_end_idx = tokenizer.apply_chat_template(
0407:                     conversation=messages[: message_idx + 1],
0408:                     tokenize=True,
0409:                     return_tensors="pt",
0410:                     padding=False,
0411:                     truncation=True,
0412:                     max_length=max_seq_length,
0413:                     add_generation_prompt=True,
0414:                 ).shape[1]
0415:             else:
0416:                 # for the last message or the message that doesn't follow with an assistant message,
0417:                 # we don't need to add the assistant generation prefix
0418:                 message_end_idx = tokenizer.apply_chat_template(
0419:                     conversation=messages[: message_idx + 1],

0427:             # set the label to -100 for the non-assistant part
0428:             labels[:, message_start_idx:message_end_idx] = -100
0429:             if max_seq_length and message_end_idx >= max_seq_length:
0430:                 break

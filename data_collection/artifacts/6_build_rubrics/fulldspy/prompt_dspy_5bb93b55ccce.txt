## Inputs
- Question ID: `dspy_5bb93b55ccce`
- Label: `module_composition_and_runtime_interoperability`
- Question: `I passed my own usage tracker through `dspy.context(track_usage=True, usage_tracker=parent)` and launched work with `dspy.Parallel`. Each worker saw token usage, but `parent` stayed empty after the batch. Is that intentional?`
- Gold answer:
Yes. `ParallelExecutor` copies thread-local DSPy overrides into each worker, and if a `usage_tracker` is present it deep-copies that tracker per worker. The worker copies accumulate usage independently; DSPy does not merge them back into the parent object.
```python
import dspy
from dspy.utils.usage_tracker import UsageTracker

parent = UsageTracker()

def task(n):
    dspy.settings.usage_tracker.add_usage('dummy', {'total_tokens': n})
    return dspy.settings.usage_tracker.get_total_tokens()

with dspy.context(track_usage=True, usage_tracker=parent):
    results = dspy.Parallel()([(task, {'n': 1}), (task, {'n': 2})])

print(results[0]['dummy']['total_tokens'])
print(results[1]['dummy']['total_tokens'])
print(dict(parent.usage_data))
```

0012: class UsageTracker:
0013:     """Tracks LM usage data within a context."""
0014: 
0015:     def __init__(self):
0016:         # Map of LM name to list of usage entries. For example:
0017:         # {
0018:         #     "openai/gpt-4o-mini": [
0019:         #         {"prompt_tokens": 100, "completion_tokens": 200},
0020:         #         {"prompt_tokens": 300, "completion_tokens": 400},
0021:         #     ],
0022:         # }
0023:         self.usage_data = defaultdict(list)
0024: 
0025:     def _flatten_usage_entry(self, usage_entry: dict[str, Any]) -> dict[str, Any]:
0026:         result = {}
0027:         for key, value in usage_entry.items():
0028:             if isinstance(value, BaseModel):
0029:                 # Convert Pydantic models to dicts, like `PromptTokensDetailsWrapper` from litellm.
0030:                 result[key] = value.model_dump()
0031:             else:
0032:                 result[key] = value
0033:         return result
0034: 
0035:     def _merge_usage_entries(
0036:         self, usage_entry1: dict[str, Any] | None, usage_entry2: dict[str, Any] | None
0037:     ) -> dict[str, Any]:
0038:         if usage_entry1 is None or len(usage_entry1) == 0:
0039:             return dict(usage_entry2)
0040:         if usage_entry2 is None or len(usage_entry2) == 0:
0041:             return dict(usage_entry1)
0042: 
0043:         result = dict(usage_entry2)
0044:         for k, v in usage_entry1.items():
0045:             current_v = result.get(k)
0046:             if isinstance(v, dict) or isinstance(current_v, dict):
0047:                 result[k] = self._merge_usage_entries(current_v, v)
0048:             elif current_v is not None or v is not None:
0049:                 result[k] = (current_v or 0) + (v or 0)
0050:         return result
0051: 
0052:     def add_usage(self, lm: str, usage_entry: dict[str, Any]) -> None:
0053:         """Add a usage entry to the tracker."""
0054:         if len(usage_entry) > 0:
0055:             self.usage_data[lm].append(self._flatten_usage_entry(usage_entry))
0056: 
0057:     def get_total_tokens(self) -> dict[str, dict[str, Any]]:
0058:         """Calculate total tokens from all tracked usage."""
0059:         total_usage_by_lm = {}
0060:         for lm, usage_entries in self.usage_data.items():
0061:             total_usage = {}
0062:             for usage_entry in usage_entries:
0063:                 total_usage = self._merge_usage_entries(total_usage, usage_entry)
0064:             total_usage_by_lm[lm] = total_usage

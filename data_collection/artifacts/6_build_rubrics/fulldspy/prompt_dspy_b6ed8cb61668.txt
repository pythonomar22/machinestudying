import dspy
from dspy.dsp.utils.utils import dotdict
from dspy.teleprompt.bootstrap_trace import FailedPrediction, bootstrap_trace_data
from dspy.utils.dummies import DummyLM


class RawSequenceLM(DummyLM):
    def __init__(self, raw_outputs):
        super().__init__([{}], adapter=dspy.JSONAdapter())
        self.raw_outputs = iter(raw_outputs)

    def forward(self, prompt=None, messages=None, **kwargs):
        text = next(self.raw_outputs)
        choice = dotdict(message=dotdict(content=text, tool_calls=None), finish_reason="stop")
        return dotdict(
            choices=[choice],
            usage=dotdict(prompt_tokens=0, completion_tokens=0, total_tokens=0),
            model="dummy",
        )


class StringToInt(dspy.Signature):
    text: str = dspy.InputField()
    number: int = dspy.OutputField()


program = dspy.Predict(StringToInt)
dataset = [
    dspy.Example(text="one", number=1).with_inputs("text"),
    dspy.Example(text="two", number=2).with_inputs("text"),
]

def exact_match(example, prediction, trace=None):
    return example.number == prediction.number


lm = RawSequenceLM(['{"number": 1}', 'not json at all'])
dspy.configure(lm=lm, adapter=dspy.JSONAdapter())
results = bootstrap_trace_data(
    program=program,
    dataset=dataset,
    metric=exact_match,
    num_threads=1,
    raise_on_error=False,
    capture_failed_parses=True,
)

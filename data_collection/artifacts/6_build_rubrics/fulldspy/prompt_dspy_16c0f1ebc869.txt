0036: def f1_score(precision, recall):
0037:     """Compute the F1 score from precision and recall, clamping both to [0, 1]."""
0038:     precision, recall = max(0.0, min(1.0, precision)), max(0.0, min(1.0, recall))
0039:     return 0.0 if precision + recall == 0 else 2 * (precision * recall) / (precision + recall)

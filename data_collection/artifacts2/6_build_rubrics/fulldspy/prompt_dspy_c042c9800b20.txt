import tempfile
import numpy as np
from dspy.retrievers.embeddings import Embeddings

def embedder(texts):
    rows = []
    for t in texts:
        if 'cat' in t:
            rows.append(np.array([1.0, 0.0, 0.0], dtype=np.float32))
        elif 'dog' in t:
            rows.append(np.array([0.0, 1.0, 0.0], dtype=np.float32))
        else:
            rows.append(np.array([0.0, 0.0, 1.0], dtype=np.float32))
    return np.stack(rows)

corpus = ['The cat sat on the mat.', 'The dog barked at the mailman.', 'Birds fly in the sky.']
orig = Embeddings(corpus=corpus, embedder=embedder, k=2, normalize=False, brute_force_threshold=1000)
with tempfile.TemporaryDirectory() as tmpdir:
    path = os.path.join(tmpdir, 'saved')
    orig.save(path)
    fresh = Embeddings(corpus=['dummy'], embedder=embedder, k=1, normalize=True, brute_force_threshold=500)
    fresh.load(path, embedder)
    a = orig('cat question')
    b = fresh('cat question')
    assert fresh.k == 2
    assert fresh.normalize is False
    assert fresh.corpus == corpus
    assert b.passages == a.passages
    assert b.indices == a.indices
    print({'k': fresh.k, 'normalize': fresh.normalize, 'corpus': fresh.corpus})

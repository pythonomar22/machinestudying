import dspy
from dspy import Example
from dspy.teleprompt import BootstrapFewShotWithRandomSearch
from dspy.utils.dummies import DummyLM


class SimpleModule(dspy.Module):
    def __init__(self):
        self.predictor = dspy.Predict("input -> output")

    def forward(self, **kwargs):
        return self.predictor(**kwargs)


trainset = [
    Example(input="q1", output="a1").with_inputs("input"),
    Example(input="q2", output="a2").with_inputs("input"),
]
student = SimpleModule(); student.set_lm(DummyLM({"q1": {"output": "a1"}, "q2": {"output": "a2"}}))
seen_inputs = []


def recording_metric(example, prediction, trace=None):
    seen_inputs.append(example.input)
    return 1.0


optimizer = BootstrapFewShotWithRandomSearch(
    metric=recording_metric,
    max_bootstrapped_demos=1,
    max_labeled_demos=0,
    num_candidate_programs=1,
    num_threads=1,
)
compiled = optimizer.compile(student, trainset=trainset, valset=None, restrict={-3})
print("valset_is_trainset:", optimizer.valset is trainset)
print("inputs_seen_by_metric:", seen_inputs)
print("candidate_seeds:", [candidate["seed"] for candidate in compiled.candidate_programs])
assert optimizer.valset is trainset
assert seen_inputs == ["q1", "q2"]
assert [candidate["seed"] for candidate in compiled.candidate_programs] == [-3]

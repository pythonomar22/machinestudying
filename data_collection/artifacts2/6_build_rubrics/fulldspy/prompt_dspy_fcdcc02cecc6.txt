import dspy
from dspy.utils.dummies import DummyLM


class UserFirstAdapter(dspy.ChatAdapter):
    def format(self, signature, demos, inputs):
        messages = super().format(signature, demos, inputs)
        return [messages[1], messages[0], *messages[2:]]


class QA(dspy.Signature):
    question: str = dspy.InputField()
    answer: str = dspy.OutputField()


adapter = UserFirstAdapter()
lm = DummyLM([{"answer": "Paris"}], adapter=adapter)
dspy.configure(lm=lm, adapter=adapter)

predict = dspy.Predict(QA)
result = predict(question="What is the capital of France?")
roles = [message["role"] for message in lm.history[-1]["messages"]]

print("ROLES:", roles)

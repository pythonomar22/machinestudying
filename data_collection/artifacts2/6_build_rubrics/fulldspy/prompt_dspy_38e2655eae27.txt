import dspy
from dspy.utils.dummies import DummyLM


class AsyncProgram(dspy.Module):
    def __init__(self):
        self.predict = dspy.Predict("question -> answer")

    async def aforward(self, question: str):
        return await self.predict.acall(question=question)


def metric(example, prediction, trace=None):
    return True


dspy.configure(lm=DummyLM([{"answer": "dummy"} for _ in range(100)]))

compiled_wrapped = optimizer.compile(dspy.syncify(AsyncProgram(), in_place=False), trainset=dataset)
print("in_place_demos:", len(compiled_in_place.predictors()[0].demos))

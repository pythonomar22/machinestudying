0511: def _convert_chat_request_to_responses_request(request: dict[str, Any]):
0512:     """
0513:     Convert a chat request to a responses request
0514:     See https://platform.openai.com/docs/api-reference/responses/create for the responses API specification.
0515:     Also see https://platform.openai.com/docs/api-reference/chat/create for the chat API specification.
0516:     """
0517:     request = dict(request)
0518:     if "messages" in request:
0519:         content_blocks = []
0520:         for msg in request.pop("messages"):
0521:             c = msg.get("content")
0522:             if isinstance(c, str):
0523:                 content_blocks.append({"type": "input_text", "text": c})
0524:             elif isinstance(c, list):
0525:                 # Convert each content item from Chat API format to Responses API format
0526:                 for item in c:
0527:                     content_blocks.append(_convert_content_item_to_responses_format(item))
0528:         request["input"] = [{"role": msg.get("role", "user"), "content": content_blocks}]
0529:     # Convert `reasoning_effort` to reasoning format supported by the Responses API
0530:     if "reasoning_effort" in request:
0531:         effort = request.pop("reasoning_effort")
0532:         request["reasoning"] = {"effort": effort, "summary": "auto"}
0533: 
0534:     # Convert `response_format` to `text.format` for Responses API
0535:     if "response_format" in request:
0536:         response_format = request.pop("response_format")
0537:         if isinstance(response_format, type) and issubclass(response_format, pydantic.BaseModel):
0538:             response_format = {
0539:                 "name": response_format.__name__,
0540:                 "type": "json_schema",
0541:                 "schema": response_format.model_json_schema(),
0542:             }
0543:         text = request.pop("text", {})
0544:         request["text"] = {**text, "format": response_format}
0545: 
0546:     return request
0547: 
0548: 
0549: def _convert_content_item_to_responses_format(item: dict[str, Any]) -> dict[str, Any]:
0550:     """
0551:     Convert a content item from Chat API format to Responses API format.
0552: 
0553:     For images, converts from:
0554:         {"type": "image_url", "image_url": {"url": "..."}}
0555:     To:
0556:         {"type": "input_image", "image_url": "..."}
0557: 
0558:     For text, converts from:
0559:         {"type": "text", "text": "..."}
0560:     To:
0561:         {"type": "input_text", "text": "..."}
0562: 
0563:     For other types, passes through as-is.
0564:     """
0565:     if item.get("type") == "image_url":
0566:         image_url = item.get("image_url", {}).get("url", "")
0567:         return {
0568:             "type": "input_image",
0569:             "image_url": image_url,
0570:         }
0571:     elif item.get("type") == "text":
0572:         return {
0573:             "type": "input_text",
0574:             "text": item.get("text", ""),
0575:         }
0576:     elif item.get("type") == "file":
0577:         file = item.get("file", {})
0578:         return {
0579:             "type": "input_file",
0580:             "file_data": file.get("file_data"),
0581:             "filename": file.get("filename"),
0582:             "file_id": file.get("file_id"),
0583:         }
0584: 
0585:     # For other items, return as-is
0586:     return item

from typing import Any

import pydantic
import dspy
from dspy.utils.dummies import DummyLM


class SchemaAwareDummy(DummyLM):
    def __init__(self, answers, adapter):
        super().__init__(answers, adapter=adapter)

    @property
    def supported_params(self):
        return {"response_format"}

    @property
    def supports_response_schema(self):
        return True


class Tight(dspy.Signature):
    question: str = dspy.InputField()
    answer: str = dspy.OutputField()


class Loose(dspy.Signature):
    question: str = dspy.InputField()
    data: dict[str, Any] = dspy.OutputField()


adapter = dspy.JSONAdapter()

lm_tight = SchemaAwareDummy([{"answer": "ok"}], adapter=adapter)
dspy.configure(lm=lm_tight, adapter=adapter)
dspy.Predict(Tight)(question="hi")
tight_format = lm_tight.history[-1]["kwargs"]["response_format"]

lm_loose = SchemaAwareDummy([{"data": {"x": 1}}], adapter=adapter)
dspy.configure(lm=lm_loose, adapter=adapter)
dspy.Predict(Loose)(question="hi")
loose_format = lm_loose.history[-1]["kwargs"]["response_format"]

print("TIGHT RESPONSE FORMAT:", tight_format)
print("LOOSE RESPONSE FORMAT:", loose_format)

assert isinstance(tight_format, type)
assert issubclass(tight_format, pydantic.BaseModel)
assert loose_format == {"type": "json_object"}

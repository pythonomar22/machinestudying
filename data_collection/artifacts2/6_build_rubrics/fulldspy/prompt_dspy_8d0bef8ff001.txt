from dspy.teleprompt import BootstrapFinetune
from dspy.utils.dummies import DummyLM


class TwoPredictorModule(dspy.Module):

        self.first = dspy.Predict("input -> mid")

        self.first.lm = lm
        self.second.lm = lm

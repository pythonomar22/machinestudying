0607: def _parse_signature(signature: str, names=None) -> dict[str, tuple[type, Any]]:
0608:     if signature.count("->") != 1:
0609:         raise ValueError(f"Invalid signature format: '{signature}', must contain exactly one '->'.")

0614:     output_fields = list(_parse_field_string(outputs_str, names))
0615:     duplicate_field_names = sorted({field_name for field_name, *_ in input_fields}.intersection(
0616:         field_name for field_name, *_ in output_fields

0618:     if duplicate_field_names:
0619:         raise ValueError(
0620:             "Input and output fields must have distinct names, but found duplicates: "
0621:             f"'{', '.join(duplicate_field_names)}'."

0625:     for field_name, field_type, is_type_undefined in input_fields:
0626:         fields[field_name] = (field_type, InputField(IS_TYPE_UNDEFINED= is_type_undefined))
0627:     for field_name, field_type, _ in output_fields:
0628:         fields[field_name] = (field_type, OutputField())

0648: def _parse_type_node(node, names=None) -> Any:
0649:     """Recursively parse an AST node representing a type annotation.
0650: 
0651:     This function converts Python's Abstract Syntax Tree (AST) nodes into actual Python types.
0652:     It's used to parse type annotations in signature strings like "x: list[int] -> y: str".
0653: 
0654:     Examples:
0655:         - For "x: int", the AST node represents 'int' and returns the int type
0656:         - For "x: list[str]", it processes a subscript node to return typing.list[str]
0657:         - For "x: Optional[int]", it handles the Union type to return Optional[int]
0658:         - For "x: MyModule.CustomType", it processes attribute access to return the actual type
0659: 
0660:     Args:
0661:         node: An AST node from Python's ast module, representing a type annotation.
0662:             Common node types include:
0663:             - ast.Name: Simple types like 'int', 'str'
0664:             - ast.Attribute: Nested types like 'typing.List'
0665:             - ast.Subscript: Generic types like 'list[int]'
0666:         names: Optional dictionary mapping type names to their actual type objects.
0667:             Defaults to Python's typing module contents plus NoneType.
0668: 
0669:     Returns:
0670:         The actual Python type represented by the AST node.
0671: 
0672:     Raises:
0673:         ValueError: If the AST node represents an unknown or invalid type annotation.
0674:     """
0675: 
0676:     if names is None:
0677:         names = dict(typing.__dict__)
0678:         names["NoneType"] = type(None)
0679: 
0680:     def resolve_name(type_name: str):
0681:         # Check if it's a built-in known type or in the provided names
0682:         if type_name in names:
0683:             return names[type_name]
0684:         # Common built-in types
0685:         builtin_types = [int, str, float, bool, list, tuple, dict, set, frozenset, complex, bytes, bytearray]
0686: 
0687:         # Check if it matches any known built-in type by name
0688:         for t in builtin_types:
0689:             if t.__name__ == type_name:
0690:                 return t
0691: 
0692:         # Attempt to import a module with this name dynamically
0693:         # This allows handling of module-based annotations like `dspy.Image`.
0694:         try:
0695:             mod = importlib.import_module(type_name)
0696:             names[type_name] = mod
0697:             return mod
0698:         except ImportError:
0699:             pass
0700: 
0701:         # If we don't know the type or module, raise an error
0702:         raise ValueError(f"Unknown name: {type_name}")
0703: 
0704:     if isinstance(node, ast.Module):
0705:         if len(node.body) != 1:
0706:             raise ValueError(f"Code is not syntactically valid: {ast.dump(node)}")
0707:         return _parse_type_node(node.body[0], names)
0708: 
0709:     if isinstance(node, ast.Expr):
0710:         return _parse_type_node(node.value, names)
0711: 
0712:     if isinstance(node, ast.Name):
0713:         return resolve_name(node.id)
0714: 
0715:     if isinstance(node, ast.Attribute):
0716:         base = _parse_type_node(node.value, names)
0717:         attr_name = node.attr
0718: 
0719:         if hasattr(base, attr_name):
0720:             return getattr(base, attr_name)
0721: 
0722:         if isinstance(node.value, ast.Name):
0723:             full_name = f"{node.value.id}.{attr_name}"
0724:             if full_name in names:
0725:                 return names[full_name]
0726: 
0727:         raise ValueError(f"Unknown attribute: {attr_name} on {base}")
0728: 
0729:     if isinstance(node, ast.Subscript):

0731:         slice_node = node.slice
0732:         if isinstance(slice_node, ast.Index):  # For older Python versions
0733:             slice_node = slice_node.value
0734: 
0735:         if isinstance(slice_node, ast.Tuple):

0737:         else:
0738:             arg_types = (_parse_type_node(slice_node, names),)
0739: 
0740:         # Special handling for Union, Optional
0741:         if base_type is typing.Union:
0742:             return typing.Union[arg_types]
0743:         if base_type is typing.Optional:
0744:             if len(arg_types) != 1:
0745:                 raise ValueError("Optional must have exactly one type argument")
0746:             return typing.Optional[arg_types[0]]
0747: 
0748:         return base_type[arg_types]
0749: 
0750:     if isinstance(node, ast.BinOp) and isinstance(node.op, ast.BitOr):
0751:         # Handle PEP 604: int | None, str | float, etc.
0752:         left = _parse_type_node(node.left, names)

0761: 
0762:     if isinstance(node, ast.Tuple):

0765:     if isinstance(node, ast.Constant):

0784: def infer_prefix(attribute_name: str) -> str:
0785:     """Infer a prefix from an attribute name by converting it to a human-readable format.
0786: 
0787:     Examples:
0788:         "camelCaseText" -> "Camel Case Text"
0789:         "snake_case_text" -> "Snake Case Text"
0790:         "text2number" -> "Text 2 Number"
0791:         "HTMLParser" -> "HTML Parser"
0792:     """
0793:     # Step 1: Convert camelCase to snake_case
0794:     # Example: "camelCase" -> "camel_Case"
0795:     s1 = re.sub("(.)([A-Z][a-z]+)", r"\1_\2", attribute_name)
0796: 
0797:     # Handle consecutive capitals
0798:     # Example: "camel_Case" -> "camel_case"
0799:     intermediate_name = re.sub("([a-z0-9])([A-Z])", r"\1_\2", s1)
0800: 
0801:     # Step 2: Handle numbers by adding underscores around them
0802:     # Example: "text2number" -> "text_2_number"
0803:     with_underscores_around_numbers = re.sub(
0804:         r"([a-zA-Z])(\d)",  # Match letter followed by number
0805:         r"\1_\2",  # Add underscore between them
0806:         intermediate_name,
0807:     )
0808:     # Example: "2text" -> "2_text"
0809:     with_underscores_around_numbers = re.sub(
0810:         r"(\d)([a-zA-Z])",  # Match number followed by letter
0811:         r"\1_\2",  # Add underscore between them
0812:         with_underscores_around_numbers,
0813:     )
0814: 
0815:     # Step 3: Convert to Title Case while preserving acronyms
0816:     words = with_underscores_around_numbers.split("_")
0817:     title_cased_words = []
0818:     for word in words:
0819:         if word.isupper():
0820:             # Preserve acronyms like 'HTML', 'API' as-is
0821:             title_cased_words.append(word)
0822:         else:
0823:             # Capitalize first letter: 'text' -> 'Text'
0824:             title_cased_words.append(word.capitalize())
0825: 
0826:     # Join words with spaces
0827:     # Example: ["Text", "2", "Number"] -> "Text 2 Number"
0828:     return " ".join(title_cased_words)

0231:         multiturn messages as expected by the LM. For custom adapters, this method can be overridden to customize
0232:         the formatting of the input messages.

0247:             {"role": "assistant", "content": conversation_history_1_output},

0279:             )

0287:             content = self.format_user_message_content(signature_without_history, inputs_copy, main_request=True)
0288:             messages.extend(conversation_history)
0289:             messages.append({"role": "user", "content": content})
0290:         else:

0292:             content = self.format_user_message_content(signature, inputs_copy, main_request=True)
0293:             messages.append({"role": "user", "content": content})
0294:

```python
import asyncio
import time

import dspy
from dspy.utils.asyncify import get_limiter


def sleeper(delay):
    time.sleep(delay)
    return 42


# asyncify consults get_limiter() on each call, so context changes retune the shared limiter.
async_sleeper = dspy.asyncify(sleeper)


async def timed_run(workers, tasks, delay):
    with dspy.context(async_max_workers=workers):
        limiter = get_limiter()
        print(f"workers={workers} total_tokens={limiter.total_tokens}")
        assert limiter.total_tokens == workers
        start = time.perf_counter()
        results = await asyncio.gather(*[async_sleeper(delay) for _ in range(tasks)])
        elapsed = time.perf_counter() - start
    return limiter, results, elapsed


async def main():
    serial_limiter, slow_results, slow_elapsed = await timed_run(1, 8, 0.2)
    parallel_limiter, fast_results, fast_elapsed = await timed_run(4, 8, 0.2)

    print("limiter_is_shared:", serial_limiter is parallel_limiter)
    print("serialish_elapsed:", round(slow_elapsed, 2))
    print("parallel_elapsed:", round(fast_elapsed, 2))

    assert serial_limiter is parallel_limiter
    assert slow_results == [42] * 8
    assert fast_results == [42] * 8

    # DSPy reuses one CapacityLimiter and updates its token count from async_max_workers.

asyncio.run(main())

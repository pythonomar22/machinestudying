0431:     def auto_budget(
0432:         self, num_preds, num_candidates, valset_size: int, minibatch_size: int = 35, full_eval_steps: int = 5
0433:     ) -> int:
0434:         import numpy as np
0435: 
0436:         num_trials = int(max(2 * (num_preds * 2) * np.log2(num_candidates), 1.5 * num_candidates))
0437:         if num_trials < 0 or valset_size < 0 or minibatch_size < 0:
0438:             raise ValueError("num_trials, valset_size, and minibatch_size must be >= 0.")
0439:         if full_eval_steps < 1:
0440:             raise ValueError("full_eval_steps must be >= 1.")
0441: 
0442:         V = valset_size
0443:         N = num_trials
0444:         M = minibatch_size
0445:         m = full_eval_steps
0446: 
0447:         # Initial full evaluation on the default program
0448:         total = V
0449: 
0450:         # Assume upto 5 trials for bootstrapping each candidate
0451:         total += num_candidates * 5
0452: 
0453:         # N minibatch evaluations
0454:         total += N * M
0455:         if N == 0:
0456:             return total  # no periodic/full evals inside the loop
0457:         # Periodic full evals occur when trial_num % (m+1) == 0, where trial_num runs 2..N+1
0458:         periodic_fulls = (N + 1) // (m) + 1
0459:         # If 1 <= N < m, the code triggers one final full eval at the end
0460:         extra_final = 1 if N < m else 0
0461: 
0462:         total += (periodic_fulls + extra_final) * V
0463:         return total

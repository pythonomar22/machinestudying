0416: ) -> tuple[dict[str, Any], dict[str, Type], dict[str, str]]:
0417:     """Convert an input json schema to tool arguments compatible with DSPy Tool.
0418: 
0419:     Args:
0420:         schema: An input json schema describing the tool's input parameters
0421: 
0422:     Returns:
0423:         A tuple of (args, arg_types, arg_desc) for DSPy Tool definition.
0424:     """
0425:     args, arg_types, arg_desc = {}, {}, {}
0426:     properties = schema.get("properties", None)
0427:     if properties is None:
0428:         return args, arg_types, arg_desc
0429: 
0430:     required = schema.get("required", [])
0431: 
0432:     defs = schema.get("$defs", {})
0433: 
0434:     for name, prop in properties.items():
0435:         if len(defs) > 0:
0436:             prop = _resolve_json_schema_reference({"$defs": defs, **prop})
0437:         args[name] = prop
0438:         arg_types[name] = _TYPE_MAPPING.get(prop.get("type"), Any)

0440:         if name in required:
0441:             arg_desc[name] += " (Required)"
0442: 
0443:     return args, arg_types, arg_desc
